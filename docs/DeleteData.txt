$repo = app(\App\Repositories\DocumentRepository::class);

foreach ($test as $e) {
    if ($e->invoice_id !== null ||
        $e->transaction_id !== null ||
        $e->purchase_order()->exists()) continue;

    if (\DB::table('bank_transactions')
        ->where('company_id', 1)
        ->whereRaw('FIND_IN_SET(?, expense_id)', [$e->hashed_id])
        ->exists()) continue;

    foreach ($e->documents()->withTrashed()->get() as $doc)
        $repo->delete($doc);

    $e->forceDelete();
}
